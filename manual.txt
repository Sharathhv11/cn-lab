2. Implement of error correction code (like CRC). 

 

#include <stdio.h> 

#include <stdlib.h> 

#include <string.h> 

 

void xorOperation(char *dividend, char *divisor, int len){ 

    for (int i = 1; i < len; i++){ 

        if(dividend[i] == divisor[i]) 

            dividend[i] = '0'; 

        else 

            dividend[i] = '1'; 

    } 

} 

 

void crcDivision(char* data, char* generator, char* remainder){ 

    int dataLen = strlen(data); 

    int genLen = strlen(generator); 

    char temp[100]; 

    strncpy(temp, data, genLen); 

    temp[genLen] = '\0'; 

 

    int cur = genLen; 

 

    while(cur < dataLen){ 

        if(temp[0] == '1') 

            xorOperation(temp, generator, genLen); 

 

        memmove(temp, temp + 1, genLen - 1); 

        temp[genLen - 1] = data[cur]; 

        cur++; 

    } 

    if(temp[0] == '1') 



        xorOperation(temp, generator, genLen); 

    strcpy(remainder, temp + 1); 

} 

 

int main(){ 

    char data[100], generator[100]; 

    char appendedData[100], remainder[100]; 

 

    printf("Enter data bits: "); 

    scanf("%s", data); 

 

    printf("Enter generator bits: "); 

    scanf("%s", generator); 

 

    int dataLen = strlen(data); 

    int genLen = strlen(generator); 

 

    strcpy(appendedData, data); 

 

    // Append zeros 

    for(int i = 0; i < genLen - 1; i++){ 

        appendedData[dataLen + i] = '0'; 

    } 

    appendedData[dataLen + genLen - 1] = '\0'; 

 

    crcDivision(appendedData, generator, remainder); 

    printf("CRC Remainder: %s\n", remainder); 

 

    char finalData[100]; 

    strcpy(finalData, data); 

    strcat(finalData, remainder); 

 

    printf("Transmitted codeword: %s\n", finalData); 



 

    char received[100], checkRemainder[100]; 

    printf("Enter received codeword: "); 

    scanf("%s", received); 

 

    crcDivision(received, generator, checkRemainder); 

 

    if(strspn(checkRemainder, "0") == strlen(checkRemainder)) 

        printf("No error detected\n"); 

    else 

        printf("Error detected\n"); 

 

    return 0; 

}
